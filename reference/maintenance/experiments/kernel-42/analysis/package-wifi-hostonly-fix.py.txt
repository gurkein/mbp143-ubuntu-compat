#!/usr/bin/env python3
"""Include dynamic Wi-Fi helpers regardless of the current failed-probe state."""
from pathlib import Path
import hashlib
import json
import shutil
import subprocess
import tempfile

ROOT=next(p for p in Path(__file__).resolve().parents if (p/'AGENTS.md').exists())
BASE=ROOT/'experiments/kernel-42/candidates/connectivity-pr596/wifi-d0-only'
OLD=BASE/'package-v0.1-3'
OUT=BASE/'package-v0.1-4'
OUT.mkdir(exist_ok=False)
STAGE=OUT/'stage'
shutil.copytree(OLD/'stage',STAGE)
meta=json.loads((OLD/'package-validation.json').read_text())
control=STAGE/'DEBIAN/control'
control.write_text(control.read_text().replace('Version: 0.1-3','Version: 0.1-4'))
# update-initramfs calls dracut directly on this host. The old native hook
# was an unsuccessful attempt based on the separately installed mkinitramfs.
(STAGE/'usr/share/initramfs-tools/hooks/mbp143-wifi-d0-vendors').unlink()
hook=STAGE/'usr/lib/dracut/modules.d/91mbp143wifid0/module-setup.sh'
text=hook.read_text()
assert '    instmods brcmfmac-wcc brcmfmac-cyw brcmfmac-bca' in text
hook.write_text(text.replace('    instmods brcmfmac-wcc brcmfmac-cyw brcmfmac-bca',
    "    # A failed probe leaves helpers unloaded; host-only filtering must not omit them.\n    hostonly='' instmods -c brcmfmac-wcc brcmfmac-cyw brcmfmac-bca"))
subprocess.run(['bash','-n',str(hook)],check=True)
# Exercise the real dracut instmods implementation, initially in host-only mode.
# This only copies modules to a temporary directory; it never loads them.
with tempfile.TemporaryDirectory(prefix='wifi-hostonly-proof-',dir='/tmp') as tmp:
    script='''set -e
source /usr/lib/dracut/dracut-functions.sh
source "$1"
kernel=7.0.14-mbp14342p1
check
kernel=7.0.0-42-generic
if check; then exit 91; fi
kernel=7.0.14-mbp14342p1
srcmods=/lib/modules/$kernel
initdir=$2
hostonly=-h
installkernel
'''
    p=subprocess.run(['bash','-c',script,'wifi-helper-test',str(hook),tmp],capture_output=True,text=True)
    assert p.returncode==0, p.stderr[-1000:]
    for name in ['brcmfmac-wcc.ko','brcmfmac-cyw.ko','brcmfmac-bca.ko','brcmfmac.ko','brcmutil.ko']:
        assert list(Path(tmp).rglob(name)),('Actual instmods did not copy module',name)
target=STAGE/'lib/modules/7.0.14-mbp14342p1/updates/mbp143-wifi-d0/brcmfmac.ko'
assert hashlib.sha256(target.read_bytes()).hexdigest()==meta['module_sha256']
doc=STAGE/'usr/share/doc'/meta['package']/'INITRAMFS-FIX'
doc.write_text('''Version 0.1-4: packaging-only repair; driver binary unchanged.
On this host update-initramfs calls dracut directly. Version 0.1-2 added a
dracut module but instmods inherited host-only filtering, omitting unloaded
vendor helpers after failed probing. Version 0.1-3 tried an unused native
initramfs-tools hook; neither image passed verification and neither was booted.
Use hostonly='' instmods -c for all three existing vendor helpers. Test against
the real dracut copy function and unpack the final initramfs before reboot.
No forced loading, live driver reset, firmware or NVRAM change is included.
''')
deb=OUT/f"{meta['package']}_0.1-4_amd64.deb"
subprocess.run(['dpkg-deb','--root-owner-group','--build',str(STAGE),str(deb)],check=True)
p=subprocess.run(['apt-get','-s','--no-remove','--no-install-recommends','install',str(deb)],capture_output=True,text=True,check=True)
actions=[l for l in p.stdout.splitlines() if l.startswith(('Inst ','Remv ','Conf '))]
assert [l.split()[1] for l in actions if l.startswith('Inst ')]==[meta['package']]
assert not any(l.startswith('Remv ') for l in actions)
meta.update(version='0.1-4',deb=str(deb.relative_to(ROOT)),deb_sha256=hashlib.sha256(deb.read_bytes()).hexdigest(),
    APT_simulation=actions,initramfs_tools_hook=False,
    native_hook_test='superseded: not called by update-initramfs on this host',
    actual_dracut_instmods_test='all five modules copied with host-only mode initially enabled; fallback excluded')
(OUT/'package-validation.json').write_text(json.dumps(meta,indent=2)+'\n')
print(json.dumps({'version':'0.1-4','actual_dracut_copy_test':'passed','driver_binary_unchanged':True}))
