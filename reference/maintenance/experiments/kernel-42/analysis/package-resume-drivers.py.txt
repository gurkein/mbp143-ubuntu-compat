#!/usr/bin/env python3
"""Build independent signed audio/Wi-Fi DEBs and test indexes offline; never install."""
from pathlib import Path
import argparse
import hashlib
import json
import os
import shutil
import subprocess
import tempfile

ROOT=next(p for p in Path(__file__).resolve().parents if (p/'AGENTS.md').exists())
KVER='7.0.14-mbp14342p1'
FALLBACK='7.0.0-42-generic'
VERSION='0.1-1'
FLAGS='-DAPPLE_PINSENSE_FIXUP -DAPPLE_CODECS -DCONFIG_SND_HDA_RECONFIG=1 -Wno-unused-variable -Wno-unused-function -Wno-error -Wno-incompatible-pointer-types -Wno-discarded-qualifiers -Wno-implicit-function-declaration'
os.umask(0o022)

def output(args):
    return subprocess.check_output([str(a) for a in args],text=True).strip()

def digest(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def package(kind):
    if kind=='audio':
        base=ROOT/'experiments/kernel-42/candidates/audio-pr220'
        source=base/'source'
        module='snd-hda-codec-cs8409'
        binary=source/'codecs/cirrus'/ (module+'.ko')
        slug='mbp143-audio-resume'
        dependency='mbp143-trial-drivers-'+KVER+' (= 7.0.0-42.42+mbp143.1)'
        details='Conditional PR220 codec reinitialization with local error propagation'
        patches=['upstream-220-kernel7.patch','error-propagation.patch','combined-kernel7.patch']
    else:
        base=ROOT/'experiments/kernel-42/candidates/connectivity-pr596/wifi-d0-only'
        source=base/'source/drivers/net/wireless/broadcom/brcm80211/brcmfmac'
        module='brcmfmac'
        binary=source/(module+'.ko')
        slug='mbp143-wifi-d0'
        dependency='linux-image-'+KVER+' (= 7.0.0-42.42+mbp143.1)'
        details='Exact-device BCM43602 PCI D0 retention only; no other Wi-Fi changes'
        patches=[]
    pkg=slug+'-'+KVER
    out=base/'package'
    out.mkdir(exist_ok=False)
    baseline=Path(output(['modinfo','-k',KVER,'-n',module]))
    fallback=Path(output(['modinfo','-k',FALLBACK,'-n',module]))
    original_hash=digest(baseline)
    fallback_hash=digest(fallback)
    cmd=['make','-C','/lib/modules/'+KVER+'/build','M='+str(source),'-j4','modules']
    if kind=='audio':cmd.insert(-1,'CFLAGS_MODULE='+FLAGS)
    build=subprocess.run(cmd,capture_output=True,text=True)
    (out/'build.log').write_text(build.stdout+build.stderr)
    build.check_returncode()
    # Every exported companion symbol must retain the target kernel ABI.
    exports={row[1]:int(row[0],16) for line in Path('/lib/modules',KVER,'build/Module.symvers').read_text().splitlines()
             if len(row:=line.split())>=3}
    companion_exports=[]
    for line in (source/'Module.symvers').read_text().splitlines():
        row=line.split()
        if len(row)>=3:
            assert exports.get(row[1])==int(row[0],16),('Changed exported ABI',row[1])
            companion_exports.append(row[1])
    assert companion_exports
    stage=out/'stage'
    subdir='updates/'+slug
    target=stage/'lib/modules'/KVER/subdir/(module+'.ko')
    target.parent.mkdir(parents=True)
    shutil.copyfile(binary,target)
    subprocess.run(['strip','--strip-debug',str(target)],check=True)
    builddir=ROOT/'experiments/kernel-42/build'
    subprocess.run([str(builddir/'scripts/sign-file'),'sha512',str(builddir/'certs/signing_key.pem'),
                    str(builddir/'certs/signing_key.x509'),str(target)],check=True)
    fields={k:output(['modinfo','-F',k,target]) for k in ['vermagic','srcversion','signer','license']}
    assert fields['vermagic'].split()[0]==KVER and fields['signer']
    new_key=output(['modinfo','-F','sig_key',target])
    baseline_key=output(['modinfo','-F','sig_key',baseline])
    companion=Path(output(['modinfo','-k',KVER,'-n','apple-ibridge']))
    assert new_key and new_key==output(['modinfo','-F','sig_key',companion])
    if baseline_key:
        assert new_key==baseline_key
    assert fields['srcversion']!=output(['modinfo','-F','srcversion',baseline])
    imports=[line.split() for line in output(['modprobe','--show-modversions',target]).splitlines()]
    assert imports
    for crc,symbol in imports:assert exports.get(symbol)==int(crc,16),('Import ABI mismatch',symbol)
    config=stage/'usr/lib/depmod.d'/('zz-'+pkg+'.conf')
    config.parent.mkdir(parents=True)
    config.write_text('# Experimental override, this exact trial kernel only.\n'+
                      f'override {module} ^7[.]0[.]14-mbp14342p1$ {subdir}\n')
    control=stage/'DEBIAN'
    control.mkdir()
    (control/'control').write_text(f'''Package: {pkg}
Version: {VERSION}
Architecture: amd64
Maintainer: Local MacBook Trial Builder <builder@localhost>
Section: kernel
Priority: optional
Depends: {dependency}, kmod, linux-initramfs-tool
Description: Isolated {kind} resume candidate for one MacBook trial kernel
 {details}.
 Preserve original modules. Planned reboot required; no live reload or sleep.
 Candidate only: runtime behavior and suspend recovery are not verified.
''')
    hook=f'''#!/bin/sh
set -e
case "$1" in
  configure|remove|purge)
    if [ -d /lib/modules/{KVER} ]; then
      depmod -a {KVER}
      if [ -e /boot/vmlinuz-{KVER} ]; then
        update-initramfs -u -k {KVER}
      fi
    fi
    ;;
esac
exit 0
'''
    for name in ['postinst','postrm']:
        f=control/name
        f.write_text(hook)
        f.chmod(0o755)
        subprocess.run(['sh','-n',str(f)],check=True)
    doc=stage/'usr/share/doc'/pkg
    doc.mkdir(parents=True)
    shutil.copyfile(base/'README.md',doc/'README')
    if kind=='audio':
        shutil.copyfile(base/'LICENSE',doc/'copyright')
        for name in patches:shutil.copyfile(base/name,doc/name)
        shutil.copyfile(base/'provenance.json',doc/'provenance.json')
    else:
        shutil.copyfile(ROOT/'experiments/kernel-42/linux-source-7.0.0/LICENSES/deprecated/ISC',doc/'copyright')
        shutil.copyfile(base.parent/'upstream/0682-brcmfmac-bcm43602-keep-d0-on-suspend.patch',doc/'0682.patch')
    # Offline indexes with all host rules; preserve the original file path.
    with tempfile.TemporaryDirectory(prefix=slug+'-indexes-') as tmp:
        sandbox=Path(tmp)
        configs=sandbox/'configs'
        configs.mkdir()
        for folder in ['/usr/lib/depmod.d','/usr/local/lib/depmod.d','/run/depmod.d','/etc/depmod.d']:
            for f in Path(folder).glob('*.conf'):shutil.copyfile(f,configs/f.name)
        shutil.copyfile(config,configs/config.name)
        tree=sandbox/'lib/modules'/KVER
        old=tree/baseline.relative_to(Path('/lib/modules')/KVER)
        new=tree/subdir/target.name
        for src,dst in [(baseline,old),(target,new)]:
            dst.parent.mkdir(parents=True,exist_ok=True)
            shutil.copyfile(src,dst)
        for name in ['modules.order','modules.builtin','modules.builtin.modinfo']:
            f=Path('/lib/modules')/KVER/name
            if f.exists():shutil.copyfile(f,tree/name)
        subprocess.run(['depmod','-b',tmp,'-C',str(configs),KVER],capture_output=True,check=True)
        selected=Path(output(['modinfo','-b',tmp,'-k',KVER,'-n',module]))
        assert selected==new and digest(selected)==digest(target)
        new.unlink()
        (configs/config.name).unlink()
        subprocess.run(['depmod','-b',tmp,'-C',str(configs),KVER],capture_output=True,check=True)
        selected=Path(output(['modinfo','-b',tmp,'-k',KVER,'-n',module]))
        assert selected==old and digest(selected)==original_hash
    deb=out/f'{pkg}_{VERSION}_amd64.deb'
    subprocess.run(['dpkg-deb','--root-owner-group','--build',str(stage),str(deb)],check=True)
    plan=subprocess.run(['apt-get','-s','--no-remove','--no-install-recommends','install',str(deb)],
                        capture_output=True,text=True,check=True)
    actions=[line for line in plan.stdout.splitlines() if line.startswith(('Inst ','Remv ','Conf '))]
    assert [line.split()[1] for line in actions if line.startswith('Inst ')]==[pkg]
    assert not any(line.startswith('Remv ') for line in actions)
    assert Path(output(['modinfo','-k',KVER,'-n',module]))==baseline and digest(baseline)==original_hash
    assert Path(output(['modinfo','-k',FALLBACK,'-n',module]))==fallback and digest(fallback)==fallback_hash
    record={'package':pkg,'version':VERSION,'deb':str(deb.relative_to(ROOT)),'deb_sha256':digest(deb),
            'module':module,'module_sha256':digest(target),**fields,
            'baseline_has_signature':bool(baseline_key),'signature_key_matches_baseline':new_key==baseline_key,
            'signature_key_matches_known_trial_companion':True,
            'matching_import_CRCs':len(imports),'unchanged_companion_export_CRCs':len(companion_exports),
            'selected_baseline_path':str(baseline),'selected_baseline_sha256':original_hash,
            'fallback_module_path':str(fallback),'fallback_sha256':fallback_hash,
            'host_selection_and_fallback_unchanged':True,'disposable_selection_and_removal_rollback':'passed',
            'APT_simulation':actions,'only_packaged_module':target.name,'kernel_scoped_override':True,
            'global_modprobe_configuration_or_firmware_blob_added':False,
            'installed':False,'loaded':False,
            'new_PM_authorization':False}
    (out/'package-validation.json').write_text(json.dumps(record,indent=2)+'\n')
    print(json.dumps(record,indent=2))

if __name__=='__main__':
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('candidate',choices=['audio','wifi'])
    package(parser.parse_args().candidate)
